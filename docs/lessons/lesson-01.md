# មេរៀនទី១៖ ផែនទី និងដំណើរការធ្វើផែនទី

!!! info "ព័ត៌មានមេរៀន"
    **រយៈពេលបង្រៀនដែលស្នើ៖ ៣ ម៉ោង** · មេរៀនទី ១ ក្នុងចំណោម ១៥

មនុស្សគ្រប់គ្នាប្រើផែនទី៖ ផែនទីក្នុងទូរស័ព្ទដើម្បីរកផ្លូវ ផែនទីទឹកជំនន់ដែលចែករំលែកលើបណ្ដាញសង្គម ឬផែនទីក្បាលដីក្នុងប្លង់។ ប៉ុន្តែផែនទីមិនមែនជារូបថតនៃពិភពពិតទេ។ វាជាលទ្ធផលនៃការសម្រេចចិត្តជាច្រើនដែលអ្នកធ្វើផែនទីបានធ្វើ ហើយការសម្រេចចិត្តទាំងនោះកំណត់ថាអ្នកអាននឹងយល់អ្វី។

មេរៀនទីមួយនៃ **ផែនទីវិទ្យា (Cartography)** ណែនាំថាផែនទីជាអ្វី ប្រភេទផែនទី របៀបដែលផែនទីធ្វើឲ្យពិភពពិតសាមញ្ញ និងដំណើរការធ្វើផែនទីដែលសៀវភៅទាំងមូលនឹងសិក្សាម្ដងមួយជំហាន។

> **កំណត់សម្គាល់៖** ឧទាហរណ៍ក្នុងសៀវភៅនេះប្រើទិន្នន័យពិតពីសំណុំទិន្នន័យ `Cambodia` ដែលប្រើរួមជាមួយសៀវភៅទី២ ([ឧបសម្ព័ន្ធ ខ](../appendix/b-cambodia-data.md))។

---

## ១.១. គោលបំណងសិក្សា

បន្ទាប់ពីបញ្ចប់មេរៀន និស្សិតអាច៖

1. កំណត់និយមន័យផែនទី និងផែនទីវិទ្យា ហើយពន្យល់ថាហេតុអ្វីផែនទីជាគំរូ មិនមែនការថតចម្លង។
2. បែងចែកផែនទីយោងទូទៅ ផែនទីប្រធានបទ និងផែនទីគោលបំណងពិសេស។
3. ពន្យល់ដំណើរការធ្វើឲ្យអរូបី៖ ការជ្រើសរើស ការធ្វើឲ្យសាមញ្ញ និងការធ្វើនិមិត្តសញ្ញា។
4. ពន្យល់គំរូទំនាក់ទំនងផែនទី និងតួនាទីនៃអ្នកធ្វើផែនទី និងអ្នកអាន។
5. រៀបរាប់ជំហាននៃដំណើរការធ្វើផែនទី និងធាតុសំខាន់ៗនៃផែនទី។
6. វាយតម្លៃផែនទីមួយដោយប្រើបញ្ជីត្រួតពិនិត្យ។

### ចំណេះដឹងមុនត្រូវមាន

- គ្មានចំណេះដឹងមុនជាក់លាក់ទេ។ មេរៀននេះជាមេរៀនដំបូងនៃស៊េរី។
- ចំណេះដឹងភូមិវិទ្យាកម្ពុជាមូលដ្ឋាន (ខេត្ត ទន្លេ ផ្លូវជាតិ)។

## ១.២. ស្ថានភាពបើកមេរៀន៖ ផែនទីទឹកជំនន់ដែលធ្វើឲ្យមនុស្សភ័យ

នៅរដូវវស្សាមួយ រូបភាពផែនទីមួយត្រូវបានចែករំលែករាប់ពាន់ដងលើបណ្ដាញសង្គម។ វាបង្ហាញខេត្តជាច្រើនពណ៌ក្រហមដិត ជាមួយពាក្យថា «តំបន់គ្រោះថ្នាក់»។ ផែនទីនោះគ្មានតារាងសម្គាល់ គ្មានកាលបរិច្ឆេទ គ្មានប្រភព និងគ្មានមាត្រដ្ឋាន។

អ្នកភូមិក្នុងស្រុកមួយដែលមិនលិចទឹក ចាប់ផ្ដើមរៀបចំចាកចេញ។ ក្រោយមកទើបដឹងថា ផែនទីនោះបង្ហាញទឹកជំនន់ពីឆ្នាំមុន ហើយពណ៌ក្រហមមានន័យថា «ខេត្តដែលមានភូមិយ៉ាងតិចមួយលិចទឹក» មិនមែន «ខេត្តទាំងមូលលិចទឹក» ទេ។

សំណួរដែលមេរៀននេះឆ្លើយ៖

- ផែនទីជាអ្វី ហើយហេតុអ្វីផែនទីអាចបំភាន់ ទោះទិន្នន័យត្រឹមត្រូវ?
- អ្នកធ្វើផែនទីសម្រេចចិត្តអ្វីខ្លះ មុនផែនទីមួយត្រូវបានចេញផ្សាយ?
- ធាតុអ្វីខ្លះដែលផែនទីនេះខ្វះ ហើយវាធ្វើឲ្យអ្នកអានយល់ខុសដូចម្ដេច?

## ១.៣. ទ្រឹស្ដីស្នូល

### ១.៣.១. ផែនទី និងផែនទីវិទ្យា

**ផែនទី (Map)** គឺជាតំណាងក្រាហ្វិកដែលធ្វើឲ្យសាមញ្ញ នៃបាតុភូតភូមិសាស្ត្រ ដែលបង្ហាញទីតាំង និងទំនាក់ទំនងលំហរបស់វា។ **ផែនទីវិទ្យា (Cartography)** គឺជាវិទ្យាសាស្ត្រ សិល្បៈ និងបច្ចេកទេសនៃការធ្វើ ប្រើ និងសិក្សាផែនទី។

ផែនទីគ្រប់ប្រភេទមានលក្ខណៈមូលដ្ឋានបី៖

| លក្ខណៈ | អត្ថន័យ | សិក្សានៅ |
|---|---|---|
| **មាត្រដ្ឋាន** | ផែនទីតូចជាងពិភពពិត ដូច្នេះត្រូវបង្រួម ហើយវត្ថុខ្លះត្រូវលុប | មេរៀនទី៤ |
| **ចំណោល** | ផែនដីមានរាងស្វ៊ែរ តែផែនទីសំប៉ែត ដូច្នេះត្រូវមានការខូចទ្រង់ទ្រាយ | មេរៀនទី២–៣ |
| **និមិត្តសញ្ញា** | វត្ថុពិតត្រូវតំណាងដោយចំណុច បន្ទាត់ ផ្ទៃ ពណ៌ និងអក្សរ | មេរៀនទី៦–៨ |

ដោយសារលក្ខណៈទាំងបីនេះ **គ្មានផែនទីណាមួយត្រឹមត្រូវគ្រប់យ៉ាងទេ**។ ផែនទីល្អ គឺផែនទីដែលសមស្របនឹង **គោលបំណង** និង **អ្នកអាន** របស់វា។

!!! note "ផែនទីវិទ្យា និង GIS"
    សៀវភៅនេះបង្រៀនពីរបៀប **តំណាង និងទំនាក់ទំនង** ព័ត៌មានភូមិសាស្ត្រតាមរយៈផែនទី។ ការរក្សាទុក និង **វិភាគ** ទិន្នន័យ ជាខ្លឹមសារ[សៀវភៅទី២៖ មូលដ្ឋានគ្រឹះនៃ GIS](https://khgeo.github.io/gis-fundamentals/)។ វគ្គទាំងពីរប្រើកម្មវិធី QGIS ដូចគ្នា។

### ១.៣.២. ប្រភេទផែនទី

ផែនទីអាចបែងចែកតាមគោលបំណង៖

| ប្រភេទ | គោលបំណង | ឧទាហរណ៍នៅកម្ពុជា |
|---|---|---|
| **ផែនទីយោងទូទៅ (General reference)** | បង្ហាញវត្ថុច្រើនប្រភេទ ដើម្បីរកទីតាំង និងទិស | ផែនទីទីសណ្ឋាន ១:៥០ ០០០ · ផែនទីរដ្ឋបាល · ផែនទីផ្លូវ |
| **ផែនទីប្រធានបទ (Thematic)** | បង្ហាញលំនាំនៃប្រធានបទតែមួយ ឬពីរ | ដង់ស៊ីតេប្រជាជន · អត្រាអក្ខរកម្ម · វិសាលភាពទឹកជំនន់ |
| **ផែនទីគោលបំណងពិសេស** | សម្រាប់ការងារជាក់លាក់ ដែលមានស្តង់ដារផ្ទាល់ | ផែនទីសុរិយោដី (ក្បាលដី) · ផែនទីនាវាចរណ៍ · ផែនទីផ្លូវអាកាស |

ផែនទីប្រធានបទបែងចែកទៀតជា **គុណភាព** (ប្រភេទ៖ ក្រប់ដី ប្រភេទសាលា) និង **បរិមាណ** (ចំនួន៖ ប្រជាជន ទឹកភ្លៀង)។ ផែនទីប្រធានបទជាផ្នែកធំនៃសៀវភៅនេះ (មេរៀនទី៩–១២)។

### ១.៣.៣. ផែនទីជាការធ្វើឲ្យអរូបី

ពិភពពិតស្មុគស្មាញពេកសម្រាប់បង្ហាញទាំងអស់។ អ្នកធ្វើផែនទីត្រូវ **ធ្វើឲ្យអរូបី (Abstraction)** តាមបីជំហាន៖

1. **ជ្រើសរើស (Selection)**៖ យកតែវត្ថុដែលពាក់ព័ន្ធនឹងគោលបំណង។ ផែនទីផ្លូវទៅសាលា មិនត្រូវការព្រំក្បាលដីទេ។
2. **ធ្វើឲ្យសាមញ្ញ (Simplification)**៖ កាត់បន្ថយលម្អិតតាមមាត្រដ្ឋាន។ ទន្លេកោងៗក្លាយជាបន្ទាត់រលោងជាងនៅមាត្រដ្ឋានតូច។
3. **ធ្វើនិមិត្តសញ្ញា (Symbolization)**៖ តំណាងវត្ថុដោយសញ្ញាដែលអ្នកអានយល់ ជាមួយតារាងសម្គាល់។

### ១.៣.៤. គំរូទំនាក់ទំនងផែនទី

ផែនទីជាឧបករណ៍ទំនាក់ទំនង។ ព័ត៌មានធ្វើដំណើរពីពិភពពិត ទៅក្នុងគំនិតរបស់អ្នកអាន តាមរយៈ **តម្រងពីរ**៖

- **តម្រងអ្នកធ្វើផែនទី**៖ គោលបំណង ទិន្នន័យដែលមាន ចំណេះដឹង និងការសម្រេចចិត្តរចនា។
- **តម្រងអ្នកអាន**៖ បទពិសោធន៍ ការអប់រំ វប្បធម៌ និងកាលៈទេសៈពេលអាន (ទូរស័ព្ទ ឬក្រដាស ប្រញាប់ ឬស្ងប់)។

ដូច្នេះ អ្នកធ្វើផែនទីល្អ **គិតពីអ្នកអានមុន** ហើយ **សាកល្បងផែនទីជាមួយអ្នកអានពិត** មុនចេញផ្សាយ។

## ១.៤. ឧទាហរណ៍ដែលបានដោះស្រាយ៖ វាយតម្លៃ និងរចនាផែនទីទឹកជំនន់ឡើងវិញ

យើងវាយតម្លៃផែនទីទឹកជំនន់ក្នុងស្ថានភាពបើកមេរៀន ដោយប្រើបញ្ជីត្រួតពិនិត្យប្រាំ ហើយស្នើការរចនាឡើងវិញសម្រាប់ខេត្តកំពង់ឆ្នាំង ដែលវិសាលភាពទឹកជំនន់ឆ្នាំ ២០១១ គ្របដណ្ដប់ ១៦៧ ៧៥៣ ហិកតា ឬ ៣១,៧% នៃផ្ទៃខេត្ត។

### ជំហានទី១៖ គោលបំណង និងអ្នកអាន

ផែនទីដើមមិនប្រាប់ថា ចង់ឲ្យអ្នកអានធ្វើអ្វី។ ការរចនាថ្មី៖ **គោលបំណង** = ប្រាប់អ្នកភូមិថាភូមិរបស់ពួកគេស្ថិតក្នុងតំបន់លិចទឹកឆ្នាំ ២០១១ ឬទេ · **អ្នកអាន** = អ្នកភូមិ មើលលើទូរស័ព្ទ។

### ជំហានទី២៖ ជ្រើសរើស និងកម្រិតលម្អិត

ខេត្តទាំងមូលពណ៌ក្រហម ធំពេកសម្រាប់គោលបំណងនេះ។ ត្រូវបង្ហាញ **វិសាលភាពទឹកជំនន់ពិតប្រាកដ** ជាពហុកោណ ជាមួយព្រំស្រុក ផ្លូវជាតិ និងឈ្មោះភូមិធំៗ ដែលអ្នកភូមិស្គាល់។

### ជំហានទី៣៖ និមិត្តសញ្ញា

ពណ៌ខៀវសម្រាប់ទឹក ជាការរំពឹងទុកធម្មតារបស់អ្នកអាន។ ពណ៌ក្រហមមិនចាំបាច់ទេ។ ផ្ទៃដែលមិនលិច ប្រើពណ៌ប្រផេះស្រាល ដើម្បីកុំទាក់ចំណាប់អារម្មណ៍។

ផែនទីជាគំរូដែលជ្រើស ធ្វើឲ្យសាមញ្ញ និងប្រើនិមិត្តសញ្ញា។ រូបភាពផ្កាយរណបក៏ត្រូវការការបកស្រាយ មុនក្លាយជាផែនទីដែរ។

**២. «ផែនទីដែលមានលម្អិតច្រើន ល្អជាងជានិច្ច»**
លម្អិតដែលមិនពាក់ព័ន្ធនឹងគោលបំណង លាក់ព័ត៌មានសំខាន់។ ភាពច្បាស់សំខាន់ជាងភាពពេញលេញ។

**៣. «ផែនទីល្អត្រូវមានធាតុផែនទីទាំងអស់ជានិច្ច»**
ធាតុនីមួយៗត្រូវឆ្លើយសំណួររបស់អ្នកអាន។ ទិសខាងជើងលើផែនទីកម្ពុជាដែលតម្រង់ខាងជើងធម្មតា ច្រើនតែមិនចាំបាច់។

**៤. «ផែនទីធ្វើដោយកម្មវិធី គឺត្រឹមត្រូវ»**
កម្មវិធីអនុវត្តតាមការកំណត់លំនាំដើម ដែលអាចមិនសមនឹងទិន្នន័យ ឬអ្នកអាន។ ការសម្រេចចិត្តផែនទីវិទ្យានៅតែជាការងាររបស់មនុស្ស។

**៥. «ផែនទីអព្យាក្រឹតជានិច្ច»**
ការជ្រើសវត្ថុ ពណ៌ ចំណងជើង និងព្រំដែន អាចបង្ហាញទស្សនៈ។ អ្នកអានគួរសួរជានិច្ចថា អ្នកណាធ្វើផែនទីនេះ ហើយដើម្បីអ្វី។

## ១.៧. សេចក្ដីសង្ខេបមេរៀន

ផែនទីជាតំណាងក្រាហ្វិកដែលធ្វើឲ្យសាមញ្ញនៃពិភពពិត ដែលមានមាត្រដ្ឋាន ចំណោល និងនិមិត្តសញ្ញា។ ផែនទីយោងទូទៅជួយរកទីតាំង ឯផែនទីប្រធានបទបង្ហាញលំនាំនៃប្រធានបទមួយ។

អ្នកធ្វើផែនទីជ្រើសរើស ធ្វើឲ្យសាមញ្ញ និងធ្វើនិមិត្តសញ្ញា ហើយព័ត៌មានឆ្លងកាត់តម្រងរបស់អ្នកធ្វើផែនទី និងអ្នកអាន។ ដំណើរការធ្វើផែនទីចាប់ផ្ដើមពីគោលបំណង ហើយធាតុផែនទីនីមួយៗត្រូវឆ្លើយសំណួររបស់អ្នកអាន។

!!! abstract "គំនិតស្នូលនៃមេរៀន"
    គ្មានផែនទីណាត្រឹមត្រូវគ្រប់យ៉ាងទេ។ ផែនទីល្អ គឺផែនទីដែលអ្នកអានគោលដៅ យល់ពីអ្វីដែលអ្នកធ្វើផែនទីចង់និយាយ ដោយមិនយល់ខុស។

## ១.៩. សំណួររំលឹក និងវាយតម្លៃ

### ក. សំណួរយល់ដឹង

1. កំណត់និយមន័យផែនទី និងពន្យល់លក្ខណៈមូលដ្ឋានបីរបស់ផែនទី។
2. បែងចែកផែនទីយោងទូទៅ និងផែនទីប្រធានបទ ដោយលើកឧទាហរណ៍ពីកម្ពុជាមួយសម្រាប់នីមួយៗ។
3. ពន្យល់ជំហានបីនៃការធ្វើឲ្យអរូបី។
4. ក្នុងគំរូទំនាក់ទំនងផែនទី «តម្រង» ទាំងពីរជាអ្វី?
5. ធាតុផែនទីណាឆ្លើយសំណួរ «ទិន្នន័យនេះជាឆ្នាំណា?»

### ខ. សំណួរអនុវត្តគោលគំនិត

6. ជ្រើសផែនទីមួយពីកាសែត ឬបណ្ដាញសង្គម ហើយវាយតម្លៃដោយបញ្ជីត្រួតពិនិត្យប្រាំក្នុងឧទាហរណ៍។
7. សម្រាប់គោលបំណង «ណែនាំភ្ញៀវទេសចរទៅប្រាសាទនៅខេត្តសៀមរាប» ចូររាយវត្ថុដែលគួរជ្រើស និងវត្ថុដែលគួរលុប។
8. ពណ៌ក្រហមប្រើសម្រាប់ «ចំនួនសិស្សខ្ពស់» ក្នុងផែនទីសាលា។ ពន្យល់ផលប៉ះពាល់តាមគំរូទំនាក់ទំនងផែនទី ហើយស្នើពណ៌ផ្សេង។
9. ផែនទីក្បាលដី (សុរិយោដី) ខុសពីផែនទីប្រធានបទដង់ស៊ីតេប្រជាជនដូចម្ដេច ទាំងគោលបំណង និងភាពត្រឹមត្រូវដែលត្រូវការ?
10. ផែនទីមួយសម្រាប់ចែករំលែកលើទូរស័ព្ទ ត្រូវមានធាតុផែនទីអ្វីខ្លះ ហើយធាតុណាអាចលុបបាន? ពន្យល់ហេតុផល។

### គ. កិច្ចការខ្លី

ប្រមូលផែនទីពីរ៖ មួយជាផែនទីយោងទូទៅ មួយជាផែនទីប្រធានបទ ពីប្រភពកម្ពុជា (សៀវភៅ កាសែត គេហទំព័រក្រសួង ឬបណ្ដាញសង្គម)។ សម្រាប់ផែនទីនីមួយៗ សរសេរ៖ គោលបំណង និងអ្នកអានដែលអ្នកប៉ាន់ស្មាន · វត្ថុដែលត្រូវបានជ្រើស · ធាតុផែនទីដែលមាន និងខ្វះ · ចំណុចមួយដែលអាចធ្វើឲ្យអ្នកអានយល់ខុស · ការកែមួយដែលអ្នកស្នើ។
